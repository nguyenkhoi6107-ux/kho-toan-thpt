\begin{ex}
Cho các biến cố $A,B$ thỏa mãn $P(A\cap B)=\dfrac15$; $P(A\cap\overline B)=\dfrac3{10}$; $P(B)=\dfrac25$.
\choiceTF
{\True $P(\overline B)=\dfrac35$.}
{$P(A\mid B)=\dfrac13$.}
{\True $P(A\mid\overline B)=\dfrac12$.}
{\True $P(A)=\dfrac12$.}
\loigiai{a) $P(\overline B)=1-P(B)=\dfrac35$. Đúng.

b) $P(A\mid B)=\dfrac{P(A\cap B)}{P(B)}=\dfrac15:\dfrac25=\dfrac12\ne\dfrac13$. Sai.

c) $P(A\mid\overline B)=\dfrac{P(A\cap\overline B)}{P(\overline B)}=\dfrac3{10}:\dfrac35=\dfrac12$. Đúng.

d) $P(A)=P(A\cap B)+P(A\cap\overline B)=\dfrac15+\dfrac3{10}=\dfrac12$. Đúng.}
\end{ex}
