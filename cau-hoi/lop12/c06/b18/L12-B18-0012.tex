\begin{ex}
Cho các biến cố $A$, $B$ thỏa mãn $P\left(\overline A\,\overline B\right)=0{,}35$; $P(A)=0{,}25$; $P(B)=0{,}6$. Giá trị $P(A\mid B)$ bằng
\choice{$\dfrac35$}{$\dfrac25$}{\True $\dfrac13$}{$\dfrac23$}
\loigiai{Ta có $P(A\cup B)=1-P\left(\overline A\,\overline B\right)=0{,}65$ nên $P(AB)=P(A)+P(B)-P(A\cup B)=0{,}25+0{,}6-0{,}65=0{,}2$. Do đó $P(A\mid B)=\dfrac{P(AB)}{P(B)}=\dfrac{0{,}2}{0{,}6}=\dfrac13$. Chọn C.}
\end{ex}
