\begin{ex}
Trong không gian, cho điểm $O$ và ba tia $Ox$, $Oy$, $Oz$ đôi một vuông góc. Gọi $\vec i$, $\vec j$, $\vec k$ lần lượt là các vectơ đơn vị trên các tia $Ox$, $Oy$, $Oz$ (hình vẽ).
\begin{center}
\begin{tikzpicture}[scale=1.1]
  \draw[truc] (0,0)--(3,0) node[below]{$y$};
  \draw[truc] (0,0)--(0,2.2) node[left]{$z$};
  \draw[truc] (0,0)--(-1.6,-1.3) node[below left]{$x$};
  \draw[->,very thick] (0,0)--(.8,0) node[above]{$\vec j$};
  \draw[->,very thick] (0,0)--(0,.8) node[right]{$\vec k$};
  \draw[->,very thick] (0,0)--(-.55,-.45) node[above left]{$\vec i$};
  \path (0,0) node[below right]{$O$};
\end{tikzpicture}
\end{center}
Mệnh đề nào sau đây sai?
\choice{$\vec i=(1;0;0)$}{\True $\vec j=(0;1;1)$}{$\vec k=(0;0;1)$}{$\vec j=(0;1;0)$}
\loigiai{$\vec i=(1;0;0)$, $\vec j=(0;1;0)$, $\vec k=(0;0;1)$ nên $\vec j=(0;1;1)$ sai.}
\end{ex}
