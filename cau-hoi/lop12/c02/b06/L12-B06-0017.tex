\begin{ex}
Cho hình hộp $ABCD.A'B'C'D'$. Đặt $\overrightarrow{AA'}=\vec a$, $\overrightarrow{AB}=\vec b$, $\overrightarrow{AD}=\vec c$. Gọi $I$ là tâm của đáy $A'B'C'D'$. Mệnh đề nào sau đây là đúng?
\begin{center}
\begin{tikzpicture}[scale=.85]
  \path (0,0) coordinate (B) (3,0) coordinate (C) (4,1.1) coordinate (D) (1,1.1) coordinate (A);
  \foreach \P in {A,B,C,D} \path (\P)+(.4,2.2) coordinate (\P');
  \coordinate (I) at ($(A')!.5!(C')$);
  \draw (B)--(C)--(D) (B')--(C')--(D')--(A')--cycle (B)--(B') (C)--(C') (D)--(D') (B)--(I);
  \draw[khuat] (B)--(A)--(D) (A)--(A');
  \draw[khuat] (A')--(C') (B')--(D');
  \fill (I) circle (1.2pt);
  \path (A) node[above left]{$A$} (B) node[below left]{$B$} (C) node[below right]{$C$} (D) node[right]{$D$}
        (A') node[above left]{$A'$} (B') node[above left]{$B'$} (C') node[above right]{$C'$} (D') node[above right]{$D'$} (I) node[above]{$I$};
\end{tikzpicture}
\end{center}
\choice{$\overrightarrow{BI}=\vec a-\dfrac12\vec c$}{$\overrightarrow{BI}=-\vec a+\vec b+\dfrac12\vec c$}{$\overrightarrow{BI}=-\vec a+2\vec b+\vec c$}{\True $\overrightarrow{BI}=\vec a-\dfrac12\vec b+\dfrac12\vec c$}
\loigiai{Vì $I$ là trung điểm của $A'C'$ nên $\overrightarrow{BI}=\overrightarrow{BB'}+\overrightarrow{B'I}=\overrightarrow{BB'}+\dfrac12\overrightarrow{B'D'}$. Ta có $\overrightarrow{BB'}=\vec a$ và $\overrightarrow{B'D'}=\overrightarrow{BD}=\overrightarrow{AD}-\overrightarrow{AB}=\vec c-\vec b$. Do đó $\overrightarrow{BI}=\vec a+\dfrac12(\vec c-\vec b)=\vec a-\dfrac12\vec b+\dfrac12\vec c$. Chọn D.}
\end{ex}
