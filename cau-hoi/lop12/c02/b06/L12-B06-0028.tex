  \coordinate (A1) at (0,1.8); \coordinate (B1) at (2,1.4); \coordinate (D1) at (.8,2.5); \coordinate (C1) at (2.8,2.1);
  \coordinate (K) at ($(C)!.5!(C1)$);

  \draw (A)--(B)--(C)--(C1)--(B1)--(A1)--cycle (B)--(B1) (C1)--(D1)--(A1);
  \fill (K) circle (1.2pt) node[right]{$K$};
  \path (A) node[left]{$A$} (B) node[below]{$B$} (C) node[right]{$C$} (D) node[above left,xshift=1pt,yshift=-1pt]{$D$}
        (A1) node[left]{$A_1$} (B1) node[below left,xshift=1pt]{$B_1$} (C1) node[above right]{$C_1$} (D1) node[above]{$D_1$};
\end{tikzpicture}
