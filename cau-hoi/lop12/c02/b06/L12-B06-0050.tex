\begin{ex}
Cho hình lập phương $ABCD.A'B'C'D'$ như hình vẽ sau.
\begin{center}
\begin{tikzpicture}[scale=2.2,x={(1cm,0cm)},y={(.55cm,.38cm)},z={(0cm,1cm)}]
  \path (0,0,0) coordinate (P1) (1,0,0) coordinate (P2) (1,1,0) coordinate (P3) (0,1,0) coordinate (P4)
        (0,0,1) coordinate (Q1) (1,0,1) coordinate (Q2) (1,1,1) coordinate (Q3) (0,1,1) coordinate (Q4);
  \draw (P1)--(P2)--(P3)--(Q3)--(Q4)--(Q1)--cycle (Q1)--(Q2)--(Q3) (P2)--(Q2);
  \draw[khuat] (P1)--(P4)--(P3) (P4)--(Q4);
  
  \path (P1) node[below left]{$B$} (P2) node[below right]{$C$} (P3) node[right]{$D$} (P4) node[left]{$A$}
        (Q1) node[left]{$B'$} (Q2) node[below left]{$C'$} (Q3) node[above right]{$D'$} (Q4) node[above left]{$A'$};
\end{tikzpicture}
\end{center}
Mệnh đề nào dưới đây đúng?
\choice{$\overrightarrow{AB'}=\overrightarrow{C'D}$}{$\overrightarrow{A'D}=\overrightarrow{CB'}$}{\True $\overrightarrow{A'B}+\overrightarrow{AD}=\overrightarrow{A'C}$}{$\overrightarrow{CA'}+\overrightarrow{CB}=\overrightarrow{DA}+\overrightarrow{A'C}$}
\loigiai{$\overrightarrow{AD}=\overrightarrow{BC}$ nên $\overrightarrow{A'B}+\overrightarrow{AD}=\overrightarrow{A'B}+\overrightarrow{BC}=\overrightarrow{A'C}$.

A sai vì $\overrightarrow{C'D}=-\overrightarrow{AB'}$; B sai vì $\overrightarrow{CB'}=-\overrightarrow{A'D}$; D sai vì $\overrightarrow{CA'}+\overrightarrow{CB}=\overrightarrow{CA'}+\overrightarrow{DA}\ne\overrightarrow{DA}+\overrightarrow{A'C}$.}
\end{ex}
