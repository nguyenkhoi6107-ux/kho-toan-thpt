\begin{ex}
Cho hình chóp $S.ABC$ có $BC=\sqrt2$, các cạnh còn lại đều bằng $1$ (tham khảo hình bên dưới). Tính tích vô hướng $\overrightarrow{SB}\cdot\overrightarrow{AC}$.
\begin{center}
\begin{tikzpicture}[scale=.9]
  \coordinate (A) at (0,0); \coordinate (B) at (.8,-1.2); \coordinate (C) at (3.5,0); \coordinate (S) at (2.5,2.5);
  \draw[khuat] (A)--(C);
  \draw (S)--(A)--(B)--(C)--cycle (S)--(B);
  \path (A) node[left]{$A$} (B) node[below]{$B$} (C) node[right]{$C$} (S) node[above]{$S$};
\end{tikzpicture}
\end{center}
\shortans{-0,5}
\loigiai{$AB^2+AC^2=2=BC^2$ nên $\overrightarrow{AB}\cdot\overrightarrow{AC}=0$; tam giác $SAC$ đều cạnh $1$ nên $\overrightarrow{AS}\cdot\overrightarrow{AC}=\dfrac12$.

$\overrightarrow{SB}\cdot\overrightarrow{AC}=(\overrightarrow{AB}-\overrightarrow{AS})\cdot\overrightarrow{AC}=0-\dfrac12=-0{,}5$.}
\end{ex}
