\begin{ex}
Cho hình lập phương $ABCD.A'B'C'D'$ (tham khảo hình vẽ). Số đo góc giữa hai vectơ $\overrightarrow{AB'}$ và $\overrightarrow{CA}$ bằng
\begin{center}
\begin{tikzpicture}[x={(1.6cm,0cm)},y={(.8cm,.55cm)},z={(0cm,1.6cm)}]
  \coordinate (D) at (0,0,0); \coordinate (C) at (1,0,0); \coordinate (B) at (1,1,0); \coordinate (A) at (0,1,0);
  \coordinate (D') at (0,0,1); \coordinate (C') at (1,0,1); \coordinate (B') at (1,1,1); \coordinate (A') at (0,1,1);
  \draw (D)--(C)--(B) (D')--(C')--(B')--(A')--cycle (D)--(D') (C)--(C') (B)--(B');
  \draw[khuat] (B)--(A)--(D) (A)--(A');
  \path (A) node[above left,font=\footnotesize]{$A$} (B) node[right,font=\footnotesize]{$B$} (C) node[below,font=\footnotesize]{$C$} (D) node[below left,font=\footnotesize]{$D$}
        (A') node[above left,font=\footnotesize]{$A'$} (B') node[above right,font=\footnotesize]{$B'$} (C') node[below right,font=\footnotesize]{$C'$} (D') node[left,font=\footnotesize]{$D'$};
\end{tikzpicture}
\end{center}
\choice{$45^\circ$}{$60^\circ$}{$90^\circ$}{\True $120^\circ$}
\loigiai{Ba đoạn $AB'$, $B'C$, $CA$ là ba đường chéo của các mặt hình lập phương nên bằng nhau, tam giác $AB'C$ đều và $\widehat{B'AC}=60^\circ$.
Vì $\overrightarrow{CA}=-\overrightarrow{AC}$ nên $(\overrightarrow{AB'},\overrightarrow{CA})=180^\circ-(\overrightarrow{AB'},\overrightarrow{AC})=180^\circ-60^\circ=120^\circ$.}
\end{ex}
