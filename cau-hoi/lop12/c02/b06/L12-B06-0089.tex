\begin{ex}
Cho hình lăng trụ $ABC.A'B'C'$ (tham khảo hình vẽ). Vectơ nào dưới đây bằng vectơ $\overrightarrow{AA'}$?
\begin{center}
\begin{tikzpicture}[scale=.9]
  \coordinate (A) at (0,0); \coordinate (C) at (3,0); \coordinate (B) at (1.5,-1);
  \coordinate (A') at (0,2.6); \coordinate (C') at (3,2.6); \coordinate (B') at (1.5,1.6);
  \draw (A)--(B)--(C) (A')--(B')--(C')--cycle (A)--(A') (B)--(B') (C)--(C');
  \draw[khuat] (A)--(C);
  \path (A) node[left]{$A$} (B) node[below]{$B$} (C) node[right]{$C$}
        (A') node[above left]{$A'$} (B') node[below right,font=\footnotesize]{$B'$} (C') node[above right]{$C'$};
\end{tikzpicture}
\end{center}
\choice{$\overrightarrow{C'C}$}{\True $\overrightarrow{CC'}$}{$\overrightarrow{AB'}$}{$\overrightarrow{B'B}$}
\loigiai{$AA'$ và $CC'$ là hai cạnh bên của hình lăng trụ nên song song, bằng nhau và cùng hướng: $\overrightarrow{AA'}=\overrightarrow{CC'}$.
$\overrightarrow{C'C}$ và $\overrightarrow{B'B}$ là các vectơ đối của $\overrightarrow{AA'}$; $\overrightarrow{AB'}$ không cùng phương với $\overrightarrow{AA'}$.}
\end{ex}
