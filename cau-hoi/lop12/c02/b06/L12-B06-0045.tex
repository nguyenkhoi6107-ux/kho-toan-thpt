\begin{ex}
Trong không gian với hai vectơ $\vec a$, $\vec b$ khác $\vec 0$, mệnh đề nào sau đây đúng?
\choice{$\vec a\cdot\vec b=|\vec a|\cdot|\vec b|\cdot\sin\left(\vec a,\vec b\right)$}{\True $\vec a\cdot\vec b=|\vec a|\cdot|\vec b|\cdot\cos\left(\vec a,\vec b\right)$}{$\vec a\cdot\vec b=|\vec a\cdot\vec b|\cdot\cos\left(\vec a,\vec b\right)$}{$\vec a\cdot\vec b=-|\vec a|\cdot|\vec b|\cdot\cos\left(\vec a,\vec b\right)$}
\loigiai{Định nghĩa tích vô hướng: $\vec a\cdot\vec b=|\vec a|\cdot|\vec b|\cdot\cos\left(\vec a,\vec b\right)$.}
\end{ex}
