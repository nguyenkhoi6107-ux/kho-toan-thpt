\begin{ex}
Cho hình chóp $S.ABC$ có $SA=SB=SC=a$, $\widehat{ASB}=60^\circ$, $\widehat{BSC}=60^\circ$ và $\widehat{CSA}=90^\circ$. Biết rằng $\overrightarrow{SA}\cdot\left(\overrightarrow{SB}-\overrightarrow{SC}\right)=k\cdot a^2$. Giá trị của $k$ bằng bao nhiêu?
\choice{$k=0{,}25$}{\True $k=0{,}5$}{$k=-0{,}25$}{$k=-0{,}5$}
\loigiai{$\overrightarrow{SA}\cdot\overrightarrow{SB}=a^2\cos60^\circ=\dfrac{a^2}2$, $\overrightarrow{SA}\cdot\overrightarrow{SC}=a^2\cos90^\circ=0$.

Vậy $\overrightarrow{SA}\cdot\left(\overrightarrow{SB}-\overrightarrow{SC}\right)=\dfrac{a^2}2$, $k=0{,}5$.}
\end{ex}
