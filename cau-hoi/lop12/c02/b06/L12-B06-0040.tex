\begin{ex}
Cho hình hộp chữ nhật $ABCD.A'B'C'D'$ có $M$ là trung điểm của đoạn thẳng $BD$ như hình bên. Mệnh đề nào trong các mệnh đề dưới đây là mệnh đề sai?
\begin{center}
\begin{tikzpicture}[scale=.8]
  \path (0,0) coordinate (A') (4,0) coordinate (D') (5.3,1.1) coordinate (C') (1.3,1.1) coordinate (B');
  \foreach \P in {A,B,C,D} \path (\P')+(0,2.2) coordinate (\P);
  \path (B)--(D) coordinate[pos=.5] (M);
  \draw (A)--(B)--(C)--(D)--cycle (A)--(A')--(D')--(C')--(C) (D)--(D') (B)--(D);
  \draw[khuat] (A')--(B')--(C') (B')--(B);
  \fill (M) circle (1.5pt);
  \path (A) node[left]{$A$} (B) node[above]{$B$} (C) node[above right]{$C$} (D) node[below right]{$D$}
        (M) node[above right=1pt]{$M$} (A') node[below left]{$A'$} (B') node[above left]{$B'$}
        (C') node[right]{$C'$} (D') node[below right]{$D'$};
\end{tikzpicture}
\end{center}
\choice{$\overrightarrow{AM}+\overrightarrow{A'C'}=\dfrac32\overrightarrow{AC}$}{\True $\overrightarrow{C'C}+\overrightarrow{CB}+\overrightarrow{CD}=2\overrightarrow{C'M}$}{$\overrightarrow{B'M}=\overrightarrow{B'B}+\dfrac12\overrightarrow{BD}$}{$\overrightarrow{AM}=\dfrac12\left(\overrightarrow{AB}+\overrightarrow{AD}\right)$}
\loigiai{$M$ là trung điểm của $BD$ nên cũng là trung điểm của $AC$.
A: $\overrightarrow{AM}+\overrightarrow{A'C'}=\dfrac12\overrightarrow{AC}+\overrightarrow{AC}=\dfrac32\overrightarrow{AC}$, đúng.
B: $\overrightarrow{CB}+\overrightarrow{CD}=2\overrightarrow{CM}$ nên vế trái bằng $\overrightarrow{C'C}+2\overrightarrow{CM}$, còn $2\overrightarrow{C'M}=2\overrightarrow{C'C}+2\overrightarrow{CM}$; hai vế khác nhau vì $\overrightarrow{C'C}\ne\vec0$, nên B sai.
C: $\overrightarrow{B'M}=\overrightarrow{B'B}+\overrightarrow{BM}=\overrightarrow{B'B}+\dfrac12\overrightarrow{BD}$, đúng.
D: quy tắc trung điểm, đúng.}
\end{ex}
