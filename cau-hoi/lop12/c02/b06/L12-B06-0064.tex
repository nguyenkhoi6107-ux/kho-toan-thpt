\begin{ex}
Cho tứ diện đều $ABCD$ có cạnh bằng $a$. Tính góc $\left(\overrightarrow{AB},\overrightarrow{CD}\right)$.
\choice{$\left(\overrightarrow{AB},\overrightarrow{CD}\right)=60^\circ$}{\True $\left(\overrightarrow{AB},\overrightarrow{CD}\right)=90^\circ$}{$\left(\overrightarrow{AB},\overrightarrow{CD}\right)=120^\circ$}{$\left(\overrightarrow{AB},\overrightarrow{CD}\right)=180^\circ$}
\loigiai{$\overrightarrow{AB}\cdot\overrightarrow{CD}=\overrightarrow{AB}\cdot\left(\overrightarrow{AD}-\overrightarrow{AC}\right)=a^2\cos60^\circ-a^2\cos60^\circ=0$.

Vậy $\left(\overrightarrow{AB},\overrightarrow{CD}\right)=90^\circ$.}
\end{ex}
