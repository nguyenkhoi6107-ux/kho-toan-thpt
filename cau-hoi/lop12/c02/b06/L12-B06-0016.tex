\begin{ex}
Cho hình lập phương $ABCD.A'B'C'D'$ có cạnh bằng $2$ (tham khảo hình bên dưới). Độ dài của vectơ $\vec u=\overrightarrow{A'C'}-\overrightarrow{A'A}$ bằng
\begin{center}
\begin{tikzpicture}[y={(.5cm,.45cm)},z={(0cm,1cm)},scale=.9]
  \coordinate (A) at (0,2,0);\coordinate (B) at (2,2,0);\coordinate (C) at (2,0,0);\coordinate (D) at (0,0,0);
  \coordinate (A') at (0,2,2);\coordinate (B') at (2,2,2);\coordinate (C') at (2,0,2);\coordinate (D') at (0,0,2);
  \draw (D)--(C)--(C')--(D')--cycle (B)--(C) (B)--(B') (B')--(C') (A')--(B') (A')--(D');
  \draw[khuat] (A)--(B) (A)--(D) (A)--(A');
  \foreach \p/\d in {A/above left,B/right,C/below right,D/below left,A'/above left,B'/above right,C'/below right,D'/above left}
    \fill (\p) circle (1pt) node[\d,font=\footnotesize] {$\p$};
\end{tikzpicture}
\end{center}
\choice{$2\sqrt2$}{$\sqrt3$}{$2\sqrt6$}{\True $2\sqrt3$}
\loigiai{Ta có $\vec u=\overrightarrow{A'C'}-\overrightarrow{A'A}=\overrightarrow{A'C'}+\overrightarrow{AA'}=\overrightarrow{AA'}+\overrightarrow{A'C'}=\overrightarrow{AC'}$. Do đó $|\vec u|=AC'=2\sqrt3$ (đường chéo của hình lập phương cạnh $2$). Chọn D.}
\end{ex}
