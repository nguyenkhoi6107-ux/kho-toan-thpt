\begin{ex}
Cho $\vec a$ và $\vec b$ là hai vectơ cùng hướng và đều khác $\vec 0$. Mệnh đề nào sau đây đúng?
\choice{\True $\vec a\cdot\vec b=|\vec a|\cdot|\vec b|$}{$\vec a\cdot\vec b=0$}{$\vec a\cdot\vec b=-1$}{$\vec a\cdot\vec b=-|\vec a|\cdot|\vec b|$}
\loigiai{Hai vectơ cùng hướng thì $\left(\vec a,\vec b\right)=0^\circ$ nên $\vec a\cdot\vec b=|\vec a|\cdot|\vec b|$.}
\end{ex}
