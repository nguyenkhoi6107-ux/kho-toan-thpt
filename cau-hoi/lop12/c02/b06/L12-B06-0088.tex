\begin{ex}
Cho hình hộp chữ nhật $ABCD.A'B'C'D'$ có $AB=2$, $AD=4$ và $AA'=2$ (tham khảo hình vẽ). Tính $\left|\overrightarrow{AB}+\overrightarrow{AD}+\overrightarrow{AA'}\right|$.
\begin{center}
\begin{tikzpicture}[x={(.75cm,0cm)},y={(.5cm,.35cm)},z={(0cm,.75cm)}]
  \coordinate (B) at (0,0,0); \coordinate (C) at (4,0,0); \coordinate (D) at (4,2,0); \coordinate (A) at (0,2,0);
  \coordinate (B') at (0,0,2); \coordinate (C') at (4,0,2); \coordinate (D') at (4,2,2); \coordinate (A') at (0,2,2);
  \draw (B)--(C)--(D) (B')--(C')--(D')--(A')--cycle (B)--(B') (C)--(C') (D)--(D');
  \draw[khuat] (B)--(A)--(D) (A)--(A');
  \path (A) node[above left,font=\footnotesize]{$A$} (B) node[below left,font=\footnotesize]{$B$} (C) node[below,font=\footnotesize]{$C$} (D) node[right,font=\footnotesize]{$D$}
        (A') node[above left,font=\footnotesize]{$A'$} (B') node[left,font=\footnotesize]{$B'$} (C') node[below right,font=\footnotesize]{$C'$} (D') node[above right,font=\footnotesize]{$D'$};
\end{tikzpicture}
\end{center}
\choice{$2\sqrt5$}{$8$}{$24$}{\True $2\sqrt6$}
\loigiai{Theo quy tắc hình hộp, $\overrightarrow{AB}+\overrightarrow{AD}+\overrightarrow{AA'}=\overrightarrow{AC'}$.
Do $ABCD.A'B'C'D'$ là hình hộp chữ nhật nên $AC'=\sqrt{AB^2+AD^2+AA'^2}=\sqrt{4+16+4}=2\sqrt6$.}
\end{ex}
