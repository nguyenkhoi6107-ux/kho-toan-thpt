\begin{ex}
Trong không gian $Oxyz$, góc giữa hai vectơ $\vec{u}=(1;1;0)$ và $\vec{v}=(0;1;1)$ bằng
\choice{$90^\circ$}{$45^\circ$}{\True $60^\circ$}{$30^\circ$}
\loigiai{$\cos(\vec{u},\vec{v})=\dfrac{0+1+0}{\sqrt2\cdot\sqrt2}=\dfrac12$ nên $(\vec{u},\vec{v})=60^\circ$.}
\end{ex}
