\begin{ex}
Một ngôi nhà gồm hai phần. Phần thân nhà dạng hình hộp chữ nhật $ABCD.OMNK$ có chiều dài $1200$ cm, chiều rộng $900$ cm, chiều cao $450$ cm. Phần mái nhà dạng hình chóp $S.ABCD$ có các cạnh bên bằng nhau và tạo với mặt đáy góc $\alpha$ với $\tan\alpha=\dfrac15$. Chọn hệ trục tọa độ $Oxyz$ sao cho $M$ thuộc tia $Ox$, $K$ thuộc tia $Oy$, $A$ thuộc tia $Oz$ (như hình vẽ). Biết $S(a;b;c)$ (đơn vị của $a,b,c$ là centimet). Tính giá trị của biểu thức $P=a+b+c$.
\begin{center}
\begin{tikzpicture}[x={(-.25cm,-.14cm)},y={(.62cm,0cm)},z={(0cm,.75cm)}]
  \coordinate (O) at (0,0,0);\coordinate (M) at (9,0,0);\coordinate (N) at (9,12,0);\coordinate (K) at (0,12,0);
  \coordinate (A) at (0,0,4.5);\coordinate (B) at (9,0,4.5);\coordinate (C) at (9,12,4.5);\coordinate (D) at (0,12,4.5);
  \coordinate (S) at (4.5,6,6);
  \draw (M)--(N)--(K) (B)--(C)--(D) (M)--(B) (N)--(C) (K)--(D) (S)--(B) (S)--(C) (S)--(D) (B)--(C);
  \draw[khuat] (O)--(M) (O)--(K) (O)--(A) (A)--(B) (A)--(D) (S)--(A);
  \foreach \p/\d in {O/left,M/below right,N/below,K/below right,A/left,B/left,C/right,D/right,S/above}
    \fill (\p) circle (1pt) node[\d,font=\footnotesize] {$\p$};
  \draw[truc] (A)--(0,0,8.2) node[above,font=\small]{$z$};
  \draw[truc] (K)--(0,14.2,0) node[right,font=\small]{$y$};
  \draw[truc] (M)--(11,0,0) node[left,font=\small]{$x$};
  \node[font=\footnotesize,below] at ($(M)!.5!(N)$) {$1200\,\mathrm{cm}$};
  \node[font=\footnotesize,below right] at ($(N)!.5!(K)$) {$900\,\mathrm{cm}$};

\shortans{1650}
\loigiai{Từ hình vẽ: $OK=MN=1200$, $OM=NK=900$, $OA=450$. Do đó $O(0;0;0)$, $M(900;0;0)$, $K(0;1200;0)$, $A(0;0;450)$ và $C(900;1200;450)$. Gọi $I$ là tâm hình chữ nhật $ABCD$: $I(450;600;450)$.

Vì các cạnh bên bằng nhau nên $SI\perp(ABCD)$, do đó $SI\parallel Oz$ và $S(450;600;c)$. Góc giữa $SC$ và đáy là $\widehat{SCI}=\alpha$, $CI=\dfrac12AC=\dfrac12\sqrt{900^2+1200^2}=750$ nên $SI=CI\tan\alpha=150$.

Suy ra $c=450+150=600$, $S(450;600;600)$ và $P=450+600+600=1650$.}
\end{ex}
