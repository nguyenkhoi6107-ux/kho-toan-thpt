\begin{ex}
Trong không gian $Oxyz$, cho ba điểm $A(0;1;2)$, $B(2;-1;1)$ và $C(1;2;-1)$. Tọa độ hình chiếu của điểm $C$ lên đường thẳng $AB$ là
\choice{$M\left(\dfrac43;\dfrac13;\dfrac23\right)$}{$N\left(\dfrac53;\dfrac23;\dfrac43\right)$}{$V\left(\dfrac23;\dfrac43;\dfrac13\right)$}{\True $X\left(\dfrac23;\dfrac13;\dfrac53\right)$}
\loigiai{$H=A+t\overrightarrow{AB}$, $\overrightarrow{AB}=(2;-2;-1)$; $\overrightarrow{CH}\cdot\overrightarrow{AB}=0\Leftrightarrow t=\dfrac{\overrightarrow{AC}\cdot\overrightarrow{AB}}{AB^2}=\dfrac39=\dfrac13$.

$H\left(\dfrac23;\dfrac13;\dfrac53\right)$.}
\end{ex}
