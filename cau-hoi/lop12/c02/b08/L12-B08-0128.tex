\begin{ex}
Trong không gian $Oxyz$, cho hai vectơ $\vec a=(1;2;3)$, $\vec b=(0;3;1)$. Tính giá trị cosin của góc giữa hai vectơ $\vec a$, $\vec b$.
\choice{$\cos\left(\vec a,\vec b\right)=\dfrac9{\sqrt{35}}$}{\True $\cos\left(\vec a,\vec b\right)=\dfrac9{2\sqrt{35}}$}{$\cos\left(\vec a,\vec b\right)=\dfrac2{\sqrt{35}}$}{$\cos\left(\vec a,\vec b\right)=\dfrac3{\sqrt{35}}$}
\loigiai{$\cos\left(\vec a,\vec b\right)=\dfrac{0+6+3}{\sqrt{14}\cdot\sqrt{10}}=\dfrac9{2\sqrt{35}}$.}
\end{ex}
