\begin{ex}
Trong không gian $Oxyz$, cho hai điểm $A(1;3;-1)$, $B(3;-1;5)$. Tọa độ của điểm $M$ thỏa mãn $\overrightarrow{MA}=3\overrightarrow{MB}$ là
\choice{$M\left(\dfrac73;\dfrac13;3\right)$}{\True $M(4;-3;8)$}{$M\left(\dfrac53;\dfrac{13}3;1\right)$}{$M\left(\dfrac73;-\dfrac13;3\right)$}
\loigiai{$A-M=3(B-M)\Leftrightarrow M=\dfrac{3B-A}2=(4;-3;8)$.}
\end{ex}
