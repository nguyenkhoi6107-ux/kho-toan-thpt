\begin{ex}
Trong không gian $Oxyz$, góc giữa hai vectơ $\vec j$ và $\vec u=\left(1;-\sqrt3;0\right)$ bằng
\choice{$\left(\vec j,\vec u\right)=120^\circ$}{$\left(\vec j,\vec u\right)=30^\circ$}{$\left(\vec j,\vec u\right)=60^\circ$}{\True $\left(\vec j,\vec u\right)=150^\circ$}
\loigiai{$\vec j=(0;1;0)$, $\cos\left(\vec j,\vec u\right)=\dfrac{-\sqrt3}{2}\Rightarrow\left(\vec j,\vec u\right)=150^\circ$.}
\end{ex}
