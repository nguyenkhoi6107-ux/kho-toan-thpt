\begin{ex}
Trong không gian $Oxyz$, góc giữa hai vectơ $\vec a=(1;2;-2)$ và $\vec b=(-1;-1;0)$ bằng
\choice{$\left(\vec a,\vec b\right)=120^\circ$}{$\left(\vec a,\vec b\right)=45^\circ$}{$\left(\vec a,\vec b\right)=60^\circ$}{\True $\left(\vec a,\vec b\right)=135^\circ$}
\loigiai{$\cos\left(\vec a,\vec b\right)=\dfrac{-3}{3\sqrt2}=-\dfrac{\sqrt2}2\Rightarrow\left(\vec a,\vec b\right)=135^\circ$.}
\end{ex}
