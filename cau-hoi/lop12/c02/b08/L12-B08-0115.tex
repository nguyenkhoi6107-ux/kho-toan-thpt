\begin{ex}
Trong không gian $Oxyz$, cho ba điểm $A(1;3;-2)$, $B(0;-1;3)$, $C(m;n;8)$ (với $m$, $n$ là tham số). Tìm tất cả các giá trị của $m$, $n$ để ba điểm $A$, $B$, $C$ thẳng hàng.
\choice{$m=3$, $n=11$}{\True $m=-1$, $n=-5$}{$m=-1$, $n=5$}{$m=1$, $n=5$}
\loigiai{$\overrightarrow{AB}=(-1;-4;5)$, $\overrightarrow{AC}=(m-1;n-3;10)=t\overrightarrow{AB}\Rightarrow t=2$.

$m=-1$, $n=-5$.}
\end{ex}
