\begin{ex}
Trong không gian $Oxyz$, cho ba điểm $A(-1;2;4)$, $B(-1;1;4)$, $C(0;0;4)$. Tìm số đo của $\widehat{ABC}$.
\choice{$\widehat{ABC}=60^\circ$}{\True $\widehat{ABC}=135^\circ$}{$\widehat{ABC}=120^\circ$}{$\widehat{ABC}=45^\circ$}
\loigiai{$\overrightarrow{BA}=(0;1;0)$, $\overrightarrow{BC}=(1;-1;0)$, $\cos\widehat{ABC}=\dfrac{-1}{\sqrt2}\Rightarrow\widehat{ABC}=135^\circ$.}
\end{ex}
