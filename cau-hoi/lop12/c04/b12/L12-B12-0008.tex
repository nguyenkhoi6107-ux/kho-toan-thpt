\begin{ex}
Giá trị của $\displaystyle\int_0^2\left(2x-\mathrm e^x\right)\mathrm dx$ bằng
\choice{$3-\mathrm e^2$}{$3-\mathrm e$}{\True $5-\mathrm e^2$}{$5-\mathrm e$}
\loigiai{$\displaystyle\int_0^2\left(2x-\mathrm e^x\right)\mathrm dx=\left(x^2-\mathrm e^x\right)\Big|_0^2=\left(4-\mathrm e^2\right)-(0-1)=5-\mathrm e^2$.}
\end{ex}
