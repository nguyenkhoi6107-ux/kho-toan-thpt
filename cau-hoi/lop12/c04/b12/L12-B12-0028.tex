\begin{ex}
Tích phân $\displaystyle\int_0^1x(1+x)\,\mathrm dx$ bằng
\choice{\True $\left.\left(\dfrac{x^2}{2}+\dfrac{x^3}{3}\right)\right|_0^1$}{$\left.(1+2x)\right|_0^1$}{$\left.\left(x^2+\dfrac{x^3}{3}\right)\right|_0^1$}{$\displaystyle\int_0^1\left(x^2+x^3\right)\mathrm dx$}
\loigiai{$x(1+x)=x+x^2$ có một nguyên hàm là $\dfrac{x^2}{2}+\dfrac{x^3}{3}$, nên $\displaystyle\int_0^1x(1+x)\,\mathrm dx=\left.\left(\dfrac{x^2}{2}+\dfrac{x^3}{3}\right)\right|_0^1$. Chọn A.}
\end{ex}
