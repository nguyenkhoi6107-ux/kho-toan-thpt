\begin{ex}
Cho hàm số $f(x)$ liên tục và có đạo hàm trên $\mathbb R$ thỏa mãn $f(2)=16$ và $\displaystyle\int_0^2f'(x)\,\mathrm dx=12$. Giá trị của $f(0)$ bằng
\choice{$-28$}{$28$}{$-4$}{\True $4$}
\loigiai{$\displaystyle\int_0^2f'(x)\,\mathrm dx=f(2)-f(0)=12\Rightarrow f(0)=f(2)-12=16-12=4$. Chọn D.}
\end{ex}
