\begin{ex}
Tính tích phân $I=\displaystyle\int_1^2\dfrac{x-1}{x}\,\mathrm dx$.
\choice{\True $I=1-\ln2$}{$I=\dfrac74$}{$I=1+\ln2$}{$I=2\ln2$}
\loigiai{$I=\displaystyle\int_1^2\left(1-\dfrac1x\right)\mathrm dx=\left(x-\ln|x|\right)\Big|_1^2=(2-\ln2)-1=1-\ln2$. Chọn A.}
\end{ex}
