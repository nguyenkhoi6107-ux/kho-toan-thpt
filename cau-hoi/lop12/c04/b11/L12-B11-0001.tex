\begin{ex}
Nguyên hàm của hàm số $f(x)=e^x$ là
\choice{$\dfrac{e^{x+1}}{x+1}+C$}{\True $e^x+C$}{$\dfrac{e^x}{x}+C$}{$x\cdot e^{x-1}+C$}
\loigiai{Ta có $\displaystyle\int e^x\,\mathrm dx=e^x+C$.}
\end{ex}
