\begin{ex}
Nguyên hàm của hàm số $f(x)=\tan^2x$ là
\choice{$\tan x+x+C$}{$\cot x-x+C$}{\True $\tan x-x+C$}{$\cot x+x+C$}
\loigiai{$\tan^2x=\dfrac1{\cos^2x}-1$ nên $\displaystyle\int\tan^2x\,\mathrm dx=\int\Big(\dfrac1{\cos^2x}-1\Big)\mathrm dx=\tan x-x+C$.}
\end{ex}
