\begin{ex}
Họ các nguyên hàm của hàm số $f(x)=\sin x$ là
\choice{\True $F(x)=-\cos x+C$}{$F(x)=-\cot x+C$}{$F(x)=\cos x+C$}{$F(x)=\cot x+C$}
\loigiai{Vì $(-\cos x)'=\sin x$ nên $\displaystyle\int\sin x\,\mathrm dx=-\cos x+C$.}
\end{ex}
