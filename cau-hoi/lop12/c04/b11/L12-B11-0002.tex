\begin{ex}
Nguyên hàm của hàm số $f(x)=\dfrac{x-1}{x}$ là
\choice{$x+\ln x+C$}{$x-\ln x+C$}{$x+\ln|x|+C$}{\True $x-\ln|x|+C$}
\loigiai{$\displaystyle\int\frac{x-1}{x}\,\mathrm dx=\int\left(1-\frac1x\right)\mathrm dx=x-\ln|x|+C$.}
\end{ex}
