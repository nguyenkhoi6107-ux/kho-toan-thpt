\choice{$\displaystyle\int f(x)\,\mathrm dx=e^{x-2}+C$}{\True $\displaystyle\int f(x)\,\mathrm dx=e^{x}+2x+C$}{$\displaystyle\int f(x)\,\mathrm dx=e^{x}+C$}{$\displaystyle\int f(x)\,\mathrm dx=e^{x}-2x+C$}
