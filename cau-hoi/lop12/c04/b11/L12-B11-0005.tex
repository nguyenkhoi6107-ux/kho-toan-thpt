\begin{ex}
Nguyên hàm của hàm số $f(x)=2^x\cdot e^x$ là
\choice{$2^x\cdot e^x+C$}{\True $\dfrac{2^x\cdot e^x}{1+\ln2}+C$}{$\dfrac{2^x\cdot e^x}{\ln2}+C$}{$2^x\cdot e^x\cdot\ln2+C$}
\loigiai{$f(x)=(2e)^x$ nên $\displaystyle\int f(x)\,\mathrm dx=\dfrac{(2e)^x}{\ln(2e)}+C=\dfrac{2^x\cdot e^x}{1+\ln2}+C$.}
\end{ex}
