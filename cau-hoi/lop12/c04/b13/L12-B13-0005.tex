\begin{ex}
Thể tích vật thể tròn xoay khi quay hình phẳng giới hạn bởi các đường $y=\sqrt{e^x-x}$, $y=0$, $x=1$, $x=2$ xung quanh trục $Ox$ là
\choice{\True $\pi\left(e^2-e-\dfrac32\right)$}{$e^2-e-\dfrac52$}{$\pi\left(e^2-e-\dfrac52\right)$}{$e^2-e-\dfrac32$}
\loigiai{$V=\pi\displaystyle\int_1^2\left(e^x-x\right)\mathrm dx=\pi\left(e^x-\dfrac{x^2}{2}\right)\Big|_1^2=\pi\left[\left(e^2-2\right)-\left(e-\dfrac12\right)\right]=\pi\left(e^2-e-\dfrac32\right)$.}
\end{ex}
