\begin{ex}
Trong không gian $Oxyz$, mặt cầu $(x-1)^2+(y-2)^2+(z+3)^2=4$ có tâm và bán kính lần lượt là
\choice{\True $I(1;2;-3)$, $R=2$}{$I(-1;-2;3)$, $R=2$}{$I(1;2;-3)$, $R=4$}{$I(-1;-2;3)$, $R=4$}
\loigiai{Mặt cầu $(x-a)^2+(y-b)^2+(z-c)^2=R^2$ có tâm $I(a;b;c)$, bán kính $R$. Ở đây $I(1;2;-3)$, $R^2=4$ nên $R=2$. Chọn A.}
\end{ex}
