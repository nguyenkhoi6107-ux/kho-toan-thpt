\begin{ex}
Trong không gian $Oxyz$, cho hình hộp chữ nhật $ABCD.A'B'C'D'$ có tọa độ đỉnh $A(0;0;0)$, $B(2;0;0)$, $D(0;3;0)$, $A'(0;0;4)$. Khoảng cách từ điểm $A$ đến mặt phẳng $(A'BD)$ bằng bao nhiêu?
\choice{\True $d\left(A;(A'BD)\right)=\dfrac{12}{\sqrt{61}}$}{$d\left(A;(A'BD)\right)=\dfrac1{\sqrt{61}}$}{$d\left(A;(A'BD)\right)=\dfrac{11}{\sqrt{61}}$}{$d\left(A;(A'BD)\right)=\sqrt{61}$}
\loigiai{$(A'BD)$: $\dfrac x2+\dfrac y3+\dfrac z4=1\Leftrightarrow6x+4y+3z-12=0$.

$d\left(A;(A'BD)\right)=\dfrac{12}{\sqrt{36+16+9}}=\dfrac{12}{\sqrt{61}}$.}
\end{ex}
