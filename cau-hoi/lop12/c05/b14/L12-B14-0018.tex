\begin{ex}
Trong không gian $Oxyz$, cho hai vectơ $\vec a=(2;-1;3)$ và $\vec b=(2;2;1)$. Tích có hướng của $\vec a$, $\vec b$ là
\choice{$\left[\vec a,\vec b\right]=(7;-4;-6)$}{\True $\left[\vec a,\vec b\right]=(-7;4;6)$}{$\left[\vec a,\vec b\right]=(2;-3;1)$}{$\left[\vec a,\vec b\right]=(5;0;0)$}
\loigiai{$\left[\vec a,\vec b\right]=(-1-6;6-2;4+2)=(-7;4;6)$.}
\end{ex}
