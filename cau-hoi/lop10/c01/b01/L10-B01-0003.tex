\begin{ex}
Với giá trị nào của $x$ thì ``$x^2-1=0,\ x\in\mathbb N$'' là mệnh đề đúng?
\choice{\True $x=1$}{$x=-1$}{$x=\pm1$}{$x=0$}
\loigiai{$x^2-1=0\Leftrightarrow x=\pm1$. Vì $x\in\mathbb N$ nên chỉ nhận $x=1$.}
\end{ex}
