\begin{ex}
Trong các mệnh đề sau, có bao nhiêu mệnh đề đúng?
\begin{enumerate}[label=\alph*)]
\item ``$\forall x\in\mathbb R,\ x^2\ge0$''.
\item ``Tồn tại số tự nhiên là số nguyên tố''.
\item ``$\exists x\in\mathbb N,\ x$ chia hết cho $x+1$''.
\item ``$\forall n\in\mathbb N,\ n^4-n^2+1$ là hợp số''.
\end{enumerate}
\shortans{3}
\loigiai{a) Đúng. b) Đúng, chẳng hạn số $2$. c) Đúng với $x=0$ ($0$ chia hết cho $1$). d) Sai với $n=0$: $n^4-n^2+1=1$ không là hợp số. Có $3$ mệnh đề đúng.}
\end{ex}
