\begin{ex}
Xét tính đúng (sai) của các mệnh đề sau
\choiceTF
{$\forall x\in\mathbb R,\ x^3-x^2+1>0$.}
{\True $\exists n\in\mathbb N,\ n^2+3$ chia hết cho $4$.}
{$P\colon$ ``$\forall x\in\mathbb R,\ \forall y\in\mathbb R\colon x+y=1$''.}
{\True $Q\colon$ ``$\exists x\in\mathbb R,\ \exists y\in\mathbb R\colon x+y=2$''.}
\loigiai{a) Sai: $x=-1$ cho $-1-1+1=-1<0$. b) Đúng: $n=1$ cho $n^2+3=4$. c) Sai: $x=0,y=0$ có $x+y=0\ne1$. d) Đúng: $x=1,y=1$.}
\end{ex}
