\begin{ex}
Cho hai số $a=\sqrt{10}+1$, $b=\sqrt{10}-1$. Hãy chọn khẳng định đúng.
\choice{\True $(a^2+b^2)\in\mathbb N$}{$(a+b)\in\mathbb Q$}{$a^2+b^2=20$}{$a\cdot b=99$}
\loigiai{$a^2+b^2=(11+2\sqrt{10})+(11-2\sqrt{10})=22\in\mathbb N$, A đúng. $a+b=2\sqrt{10}$ vô tỉ nên B sai; $a^2+b^2=22\ne20$ nên C sai; $ab=10-1=9\ne99$ nên D sai.}
\end{ex}
