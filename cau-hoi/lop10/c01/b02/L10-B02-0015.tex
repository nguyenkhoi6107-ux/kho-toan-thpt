\begin{ex}
Cho các tập hợp
\begin{align*}
A&=\{-3;-2;-1;0;1;2;3\};\\
B&=\{0;1;4;5\};\\
C&=\{-4;-3;1;2;5;6\}.
\end{align*}
Các mệnh đề sau đúng hay sai?
\choiceTF
{\True $A\cup B=\{-3;-2;-1;0;1;2;3;4;5\}$.}
{$A\cap B=\{0\}$.}
{\True $(A\cup B)\cap C=\{-3;1;2;5\}$.}
{\True $A\cap B\cap C=\{1\}$.}
\loigiai{a) Đúng. b) $A\cap B=\{0;1\}\ne\{0\}$, sai. c) $A\cup B=\{-3;\dots;5\}$, giao với $C$ được $\{-3;1;2;5\}$. Đúng. d) $A\cap B=\{0;1\}$, giao với $C$ được $\{1\}$. Đúng.}
\end{ex}
