\begin{ex}
Cho $A=[-1;3]$; $B=(2;5)$. Tìm mệnh đề sai.
\choice{$B\setminus A=(3;5)$}{$A\cap B=(2;3]$}{$A\setminus B=[-1;2]$}{\True $A\cup B=[-1;5]$}
\loigiai{$A\cup B=[-1;5)$ vì $5\notin B$ và $5\notin A$; nên D sai. Các ý còn lại đúng: $B\setminus A=(3;5)$, $A\cap B=(2;3]$, $A\setminus B=[-1;2]$.}
\end{ex}
