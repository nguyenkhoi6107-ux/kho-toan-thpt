\begin{ex}
Liệt kê các phần tử của tập hợp $A=\{x\in\mathbb N\mid(6x^2-7x+1)(x^2-4)=0\}$ ta được
\choice{$A=\left\{\dfrac16;\dfrac12;2\right\}$}{$A=\{-2;1;2\}$}{$A=\left\{-2;\dfrac16;1;2\right\}$}{\True $A=\{1;2\}$}
\loigiai{$6x^2-7x+1=(6x-1)(x-1)=0\Leftrightarrow x\in\left\{\dfrac16;1\right\}$; $x^2-4=0\Leftrightarrow x=\pm2$. Chỉ nhận nghiệm tự nhiên: $A=\{1;2\}$.}
\end{ex}
