\begin{ex}
Cho hai tập hợp $A$ và $B$ biết $A\setminus B=\{a;f\}$, $A\cup B=\{a;b;c;d;e;f;g;h\}$, $B\setminus A=\{b;g;h\}$. Vậy:
\choiceTF
{\True $A=\{a;c;d;e;f\}$.}
{$A\subset B$.}
{\True $A\cap B=\{c;d;e\}$.}
{$A\subset B$.}
\loigiai{$A\cup B$ gồm ba phần: $A\setminus B$, $A\cap B$, $B\setminus A$. Suy ra $A\cap B=\{a;b;c;d;e;f;g;h\}\setminus(\{a;f\}\cup\{b;g;h\})=\{c;d;e\}$ và $A=(A\setminus B)\cup(A\cap B)=\{a;f;c;d;e\}$. a) Đúng. b) và d) sai vì $a\in A$ nhưng $a\notin B$. c) Đúng.}
\end{ex}
