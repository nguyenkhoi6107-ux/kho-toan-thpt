\begin{ex}
Viết tập hợp $A=\{x\in\mathbb N\mid(2x+1)(x^2-5x+6)=0\}$ bằng cách liệt kê.
\choice{$\{-1;2;3\}$}{\True $\{2;3\}$}{$\left\{-\dfrac12;2;3\right\}$}{$\{-1;2\}$}
\loigiai{$2x+1=0\Leftrightarrow x=-\dfrac12\notin\mathbb N$; $x^2-5x+6=0\Leftrightarrow x\in\{2;3\}$. Vậy $A=\{2;3\}$.}
\end{ex}
