\begin{ex}
Tập tất cả các giá trị $m$ để hàm số $y=\sqrt{x-2m}+\sqrt{m-x}$ xác định trên khoảng $(2;3)$ là
\choice{$\dfrac32\le m\le2$}{\True $m\in\varnothing$}{$m>2$}{$m\le\dfrac32$}
\loigiai{Hàm số xác định khi $2m\le x\le m$. Cần $(2;3)\subset[2m;m]$, tức là $2m\le2$ và $m\ge3$: vô lí. Vậy $m\in\varnothing$.}
\end{ex}
