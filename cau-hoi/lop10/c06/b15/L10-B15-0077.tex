\begin{ex}
Tìm giá trị của tham số $m$ để hàm số $y=\dfrac{1}{\sqrt{x^2-2x-m}}$ xác định trên $[2;3]$.
\choice{\True $m<0$}{$0<m<3$}{$m\le0$}{$m\ge3$}
\loigiai{Cần $x^2-2x>m$ với mọi $x\in[2;3]$. Trên $[2;3]$, $x^2-2x=x(x-2)$ nhỏ nhất bằng $0$ tại $x=2$. Vậy $m<0$.}
\end{ex}
