\begin{ex}
Tập xác định của hàm số $y=\dfrac{\sqrt{4-x}+\sqrt{x+2}}{x^2-x-12}$ là
\choice{$[-2;4]$}{$(-3;-2)\cup(-2;4)$}{$(-2;4)$}{\True $[-2;4)$}
\loigiai{Điều kiện $4-x\ge0$, $x+2\ge0$ và $x^2-x-12\ne0$, tức là $-2\le x\le4$, $x\ne4$, $x\ne-3$. Vậy $D=[-2;4)$.}
\end{ex}
