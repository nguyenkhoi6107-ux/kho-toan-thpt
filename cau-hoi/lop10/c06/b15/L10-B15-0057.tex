\begin{ex}
Tập xác định của hàm số $y=\dfrac{3-x}{x^2-5x-6}$ là
\choice{\True $D=\mathbb R\setminus\{-1;6\}$}{$D=\mathbb R\setminus\{1;-6\}$}{$D=\{-1;6\}$}{$D=\{1;-6\}$}
\loigiai{Điều kiện $x^2-5x-6\ne0\Leftrightarrow x\ne-1$ và $x\ne6$. Vậy $D=\mathbb R\setminus\{-1;6\}$.}
\end{ex}
