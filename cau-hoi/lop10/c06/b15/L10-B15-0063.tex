\choice{$D=[-2;+\infty)\setminus\{-4\}$}{$D=[-2;+\infty)$}{$D=\mathbb R\setminus\{-4;3\}$}{\True $D=[-2;+\infty)\setminus\{3\}$}

\end{ex}
