\begin{ex}
Tìm tập xác định $D$ của hàm số $y=\dfrac{x+1}{(x+1)(x^2-4)}$.
\choice{$D=\mathbb R\setminus\{2\}$}{$D=\mathbb R\setminus\{\pm2\}$}{$D=\mathbb R\setminus\{-1;2\}$}{\True $D=\mathbb R\setminus\{-1;\pm2\}$}
\loigiai{Điều kiện $(x+1)(x^2-4)\ne0\Leftrightarrow x\ne-1$, $x\ne\pm2$ (không rút gọn trước khi tìm điều kiện). Vậy $D=\mathbb R\setminus\{-1;\pm2\}$.}
\end{ex}
