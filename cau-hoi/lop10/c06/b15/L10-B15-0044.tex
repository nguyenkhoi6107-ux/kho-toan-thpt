\loigiai{$D=[-2;2]$. Với $x\in D$ có $0\le4-x^2\le4$ nên $0\le y\le2$; mọi $y_0\in[0;2]$ đạt được tại $x=\sqrt{4-y_0^2}$. Vậy $T=[0;2]$.}
\end{ex}
