\begin{ex}
Tìm tập xác định $D$ của hàm số $y=\sqrt{2x-9}$.
\choice{$D=\mathbb R\setminus\left\{\dfrac92\right\}$}{$D=\left(-\infty;\dfrac92\right]$}{$D=\left(\dfrac92;+\infty\right)$}{\True $D=\left[\dfrac92;+\infty\right)$}
\loigiai{Điều kiện $2x-9\ge0\Leftrightarrow x\ge\tfrac92$. Vậy $D=\left[\tfrac92;+\infty\right)$.}
\end{ex}
