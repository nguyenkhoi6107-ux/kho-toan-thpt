\begin{ex}
Tìm giá trị của tham số $m$ để hàm số $y=\dfrac{x+1}{x-2m+1}$ xác định trên nửa khoảng $(0;1]$.
\choice{$\left[\begin{array}{l}m\le\frac12\\ m\ge1\end{array}\right.$}{\True $\left[\begin{array}{l}m\le\frac12\\ m>1\end{array}\right.$}{$\left[\begin{array}{l}m<\frac12\\ m\ge1\end{array}\right.$}{$\left[\begin{array}{l}m<\frac12\\ m>1\end{array}\right.$}
\loigiai{Hàm số xác định khi $x\ne2m-1$. Cần $2m-1\notin(0;1]$, tức là $2m-1\le0$ hoặc $2m-1>1$. Vậy $m\le\tfrac12$ hoặc $m>1$.}
\end{ex}
