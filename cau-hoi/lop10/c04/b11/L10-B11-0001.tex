\begin{ex}
Cho hai vectơ $\vec a$, $\vec b$ thỏa mãn điều kiện $|\vec a|=1$, $|\vec b|=2$ và $\vec a\cdot\vec b=-1$. Khi đó $|\vec a-\vec b|$ bằng
\choice{\True $\sqrt7$}{$\sqrt5$}{$3$}{$1$}
\loigiai{$|\vec a-\vec b|^2=|\vec a|^2-2\,\vec a\cdot\vec b+|\vec b|^2=1+2+4=7$, suy ra $|\vec a-\vec b|=\sqrt7$.}
\end{ex}
