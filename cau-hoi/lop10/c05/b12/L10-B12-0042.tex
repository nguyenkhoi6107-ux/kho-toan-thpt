\begin{ex}
Một thửa ruộng hình chữ nhật có chiều dài $x=23\text{ m}\pm0{,}01\text{ m}$ và chiều rộng $y=15\text{ m}\pm0{,}01\text{ m}$. Diện tích của thửa ruộng là
\choice{$S=345\text{ m}^2\pm0{,}03801\text{ m}^2$}{$S=345\text{ m}^2\pm0{,}38\text{ m}^2$}{$S=345\text{ m}^2\pm0{,}0038\text{ m}^2$}{\True $S=345\text{ m}^2\pm0{,}3801\text{ m}^2$}
\loigiai{$S=(23+a)(15+b)=345+23b+15a+ab$ với $|a|,|b|\le0{,}01$.

$|23b+15a+ab|\le0{,}23+0{,}15+0{,}0001=0{,}3801$. Vậy $S=345\text{ m}^2\pm0{,}3801\text{ m}^2$. Chọn D.}
\end{ex}
