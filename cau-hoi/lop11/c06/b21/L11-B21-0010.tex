\begin{ex}
Tập nghiệm của bất phương trình $\left(\dfrac15\right)^{x-1}<25$ là
\choice{\True $(-1;+\infty)$}{$(-3;+\infty)$}{$(-2;+\infty)$}{$(-\infty;-1)$}
\loigiai{$\left(\dfrac15\right)^{x-1}<25\Leftrightarrow5^{1-x}<5^2\Leftrightarrow1-x<2$ (vì cơ số $5>1$) $\Leftrightarrow x>-1$. Tập nghiệm là $(-1;+\infty)$.}
\end{ex}
