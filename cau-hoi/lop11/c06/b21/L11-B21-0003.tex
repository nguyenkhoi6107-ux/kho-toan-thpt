\begin{ex}
Nghiệm của phương trình $\log_2(x-1)=3$ là
\choice{$x=10$}{$x=8$}{\True $x=9$}{$x=7$}
\loigiai{Điều kiện $x>1$. $\log_2(x-1)=3\Leftrightarrow x-1=2^3=8\Leftrightarrow x=9$.}
\end{ex}
