\begin{ex}
Nghiệm của phương trình $3^{3x-2}=9^x$ là
\choice{$x=4$}{$x=1$}{\True $x=2$}{$x=3$}
\loigiai{$3^{3x-2}=9^x\Leftrightarrow3^{3x-2}=3^{2x}\Leftrightarrow3x-2=2x\Leftrightarrow x=2$.}
\end{ex}
