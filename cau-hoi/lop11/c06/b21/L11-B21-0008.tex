\begin{ex}
Tập xác định của hàm số $y=\sqrt{\log_2x}$ là
\choice{$D=(0;1]$}{$D=[2;+\infty)$}{$D=(0;+\infty)$}{\True $D=[1;+\infty)$}
\loigiai{Hàm số xác định khi $x>0$ và $\log_2x\ge0\Leftrightarrow x\ge2^0=1$ (vì cơ số $2>1$). Vậy $D=[1;+\infty)$.}
\end{ex}
