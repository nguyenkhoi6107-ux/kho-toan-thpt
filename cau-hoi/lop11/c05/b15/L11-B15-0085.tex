\begin{ex}
Tìm giới hạn \[\lim\left[\dfrac1{1\cdot3}+\dfrac1{3\cdot5}+\dfrac1{5\cdot7}+\cdots+\dfrac1{(2n-1)(2n+1)}\right].\]
\shortans{0,5}
\loigiai{\sloppy $\dfrac1{(2k-1)(2k+1)}=\dfrac12\left(\dfrac1{2k-1}-\dfrac1{2k+1}\right)$ nên tổng $=\dfrac12\left(1-\dfrac1{2n+1}\right)\to\dfrac12=0{,}5$.}
\end{ex}
