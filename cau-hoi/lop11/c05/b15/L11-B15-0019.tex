\begin{ex}
Tính giới hạn $L=\lim\dfrac{2n+1}{2+n-n^2}$.
\choice{$L=-\infty$}{$L=-2$}{$L=1$}{\True $L=0$}
\loigiai{\sloppy Chia tử và mẫu cho $n^2$: $\dfrac{\frac2n+\frac1{n^2}}{\frac2{n^2}+\frac1n-1}\to\dfrac{0}{-1}=0$.}
\end{ex}
