\begin{ex}
Cho cấp số cộng $(u_n)$ có số hạng đầu $u_1=2$ và công sai $d=3$. Tìm $\lim\dfrac n{u_n}$.
\choice{\True $L=\dfrac13$}{$L=\dfrac12$}{$L=3$}{$L=2$}
\loigiai{\sloppy $u_n=2+3(n-1)=3n-1$, nên $\dfrac n{u_n}=\dfrac{1}{3-\frac1n}\to\dfrac13$.}
\end{ex}
