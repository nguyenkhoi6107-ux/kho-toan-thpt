\begin{ex}
Tính giới hạn $T=\lim\left(\sqrt{16^{n+1}+4^n}-\sqrt{16^{n+1}+3^n}\right)$.
\choice{$T=0$}{$T=\dfrac14$}{\True $T=\dfrac18$}{$T=\dfrac1{16}$}
\loigiai{\sloppy Nhân lượng liên hợp: $T=\lim\dfrac{4^n-3^n}{\sqrt{16^{n+1}+4^n}+\sqrt{16^{n+1}+3^n}}$. Chia tử và mẫu cho $4^n$ (chú ý $\sqrt{16^{n+1}}=4\cdot4^n$): \[T=\lim\dfrac{1-\left(\frac34\right)^n}{\sqrt{16+\left(\frac14\right)^n}+\sqrt{16+\left(\frac3{16}\right)^n}}=\dfrac1{4+4}=\dfrac18.\]}
\end{ex}
