\begin{ex}
Tìm $L=\lim\left(\dfrac11+\dfrac1{1+2}+\cdots+\dfrac1{1+2+\cdots+n}\right)$.
\choice{$L=\dfrac52$}{$L=+\infty$}{\True $L=2$}{$L=\dfrac32$}
\loigiai{\sloppy $1+2+\cdots+k=\dfrac{k(k+1)}2$ nên $\dfrac1{1+2+\cdots+k}=2\left(\dfrac1k-\dfrac1{k+1}\right)$. Tổng $=2\left(1-\dfrac1{n+1}\right)\to2$.}
\end{ex}
