\begin{ex}
Giá trị của tổng \[S=\dfrac{12}{4\cdot16}+\dfrac{20}{16\cdot36}+\dfrac{28}{36\cdot64}+\cdots+\dfrac{84}{400\cdot484}\] là
\choice{$\dfrac{31}{121}$}{\True $\dfrac{30}{121}$}{$\dfrac{32}{121}$}{$\dfrac{33}{121}$}
\loigiai{\sloppy Số hạng tổng quát $\dfrac{8k+4}{(2k)^2(2k+2)^2}=\dfrac14\left[\dfrac1{k^2}-\dfrac1{(k+1)^2}\right]$, $k=1,2,\dots,10$. Do đó $S=\dfrac14\left(1-\dfrac1{121}\right)=\dfrac{30}{121}$.}
\end{ex}
