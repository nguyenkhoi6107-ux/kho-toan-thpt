\begin{ex}
Cho dãy số $(u_n)$ biết $\begin{cases}u_1=-2\\u_n=3u_{n-1}-1,\ \forall n\ge2\end{cases}$. Tính $L=\lim\dfrac{u_n}{3^n}$.
\choice{Không xác định.}{$L=+\infty$}{\True $L=-\dfrac56$}{$L=0$}
\loigiai{\sloppy Đặt $v_n=u_n-\dfrac12$. Khi đó $v_n=3v_{n-1}$, $v_1=-\dfrac52$ nên $v_n=-\dfrac52\cdot3^{n-1}$ và $u_n=\dfrac12-\dfrac52\cdot3^{n-1}$. Vậy $\dfrac{u_n}{3^n}=\dfrac1{2\cdot3^n}-\dfrac56\to-\dfrac56$.}
\end{ex}
