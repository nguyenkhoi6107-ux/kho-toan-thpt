\begin{ex}
Trong các giới hạn sau đây, giới hạn nào có giá trị bằng $1$?
\choice{$\lim\dfrac{3^{n+1}+2n}{5+3^n}$.}{$\lim\dfrac{3n^2+n}{4n^2-5}$.}{\True $\lim\left(\sqrt{n^2+2n}-\sqrt{n^2+1}\right)$.}{$\lim\dfrac{2n^3+3}{1+2n^2}$.}
\loigiai{\sloppy A: $\to3$. B: $\to\dfrac34$. D: $\to+\infty$. C: \[\dfrac{2n-1}{\sqrt{n^2+2n}+\sqrt{n^2+1}}=\dfrac{2-\frac1n}{\sqrt{1+\frac2n}+\sqrt{1+\frac1{n^2}}}\to1.\]}
\end{ex}
