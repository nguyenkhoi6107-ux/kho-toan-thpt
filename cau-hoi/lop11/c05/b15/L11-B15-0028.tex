\begin{ex}
Cho hai dãy số $(u_n)$ và $(v_n)$ có $u_n=\dfrac1{n+1}$; $v_n=\dfrac3{n+3}$. Tính $\lim\dfrac{u_n}{v_n}$.
\choice{$0$}{$3$}{\True $\dfrac13$}{$+\infty$}
\loigiai{\sloppy \[\dfrac{u_n}{v_n}=\dfrac{n+3}{3(n+1)}=\dfrac{1+\frac3n}{3\left(1+\frac1n\right)}\to\dfrac13.\]}
\end{ex}
