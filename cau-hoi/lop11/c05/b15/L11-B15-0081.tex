\begin{ex}
Tìm giới hạn $\lim\dfrac{1+2+\cdots+n}{n^2+3n}$.
\shortans{0,5}
\loigiai{\sloppy $1+2+\cdots+n=\dfrac{n(n+1)}2$ nên biểu thức bằng $\dfrac{n(n+1)}{2n(n+3)}=\dfrac{n+1}{2(n+3)}\to\dfrac12=0{,}5$.}
\end{ex}
