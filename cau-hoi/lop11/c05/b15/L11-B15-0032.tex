\begin{ex}
Giới hạn của dãy số $(u_n)$ với $u_n=3^n-4^n$ là
\choice{$+\infty$}{\True $-\infty$}{$\dfrac43$}{$1$}
\loigiai{\sloppy $3^n-4^n=4^n\left[\left(\frac34\right)^n-1\right]$; $4^n\to+\infty$ và thừa số trong ngoặc $\to-1$ nên giới hạn bằng $-\infty$.}
\end{ex}
