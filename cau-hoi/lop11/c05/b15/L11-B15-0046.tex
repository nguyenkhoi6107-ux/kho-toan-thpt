\begin{ex}
Tính giới hạn $L=\lim\left(\sqrt{9n^2+2n-1}-\sqrt{4n^2+1}\right)$.
\choice{\True $+\infty$}{$1$}{$-\infty$}{$\dfrac94$}
\loigiai{\sloppy \[\sqrt{9n^2+2n-1}-\sqrt{4n^2+1}=n\left(\sqrt{9+\frac2n-\frac1{n^2}}-\sqrt{4+\frac1{n^2}}\right);\] $n\to+\infty$ và thừa số còn lại $\to3-2=1>0$, nên $L=+\infty$.}
\end{ex}
