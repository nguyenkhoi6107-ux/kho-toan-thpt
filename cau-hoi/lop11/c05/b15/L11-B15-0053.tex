\begin{ex}
Tính tổng $S$ của cấp số nhân lùi vô hạn có số hạng đầu $u_1=1$ và công bội $q=-\dfrac12$.
\choice{$S=2$}{$S=\dfrac32$}{$S=1$}{\True $S=\dfrac23$}
\loigiai{\sloppy $S=\dfrac{u_1}{1-q}=\dfrac1{1+\frac12}=\dfrac23$.}
\end{ex}
