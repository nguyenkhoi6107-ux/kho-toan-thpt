\begin{ex}
Cho dãy số $(u_n)$ thỏa $|u_n-2|<\dfrac1{n^3}$ với mọi $n\in\mathbb N^*$. Khi đó
\choice{$\lim u_n$ không tồn tại.}{$\lim u_n=1$.}{$\lim u_n=0$.}{\True $\lim u_n=2$.}
\loigiai{\sloppy Vì $0\le|u_n-2|<\dfrac1{n^3}\to0$ nên $\lim(u_n-2)=0$, suy ra $\lim u_n=2$.}
\end{ex}
