\begin{ex}
Cho dãy số $(u_n)$ với $u_n=an^2+n-1$ với $a\in\mathbb R$.
\choiceTF
{Với $a=1$, giới hạn của dãy số đã cho là $1$.}
{\True Với $a=2$, giới hạn của dãy số đã cho là $+\infty$.}
{\True Với $a=-\dfrac52$, giới hạn của dãy số đã cho là $-\infty$.}
{Với $a\le0$, giới hạn của dãy số đã cho là $-\infty$.}
\loigiai{\sloppy $u_n=n^2\left(a+\dfrac1n-\dfrac1{n^2}\right)$ với $n^2\to+\infty$.
a) $a=1$: $u_n\to+\infty$. Sai.
b) $a=2>0$: $u_n\to+\infty$. Đúng.
c) $a=-\dfrac52<0$: $u_n\to-\infty$. Đúng.
d) Với $a=0$: $u_n=n-1\to+\infty$. Sai.}
\end{ex}
