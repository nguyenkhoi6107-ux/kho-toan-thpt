\begin{ex}
Cho dãy số $(u_n)$ với $u_1=-2$, $u_n=3u_{n-1}-1$, $\forall n\ge2$. Tính $\lim\dfrac{u_n}{3^n}$ (kết quả làm tròn đến hàng phần mười).
\shortans{-0,8}
\loigiai{\sloppy Đặt $v_n=u_n-\dfrac12$ thì $v_n=3v_{n-1}$, $v_1=-\dfrac52$, nên $u_n=\dfrac12-\dfrac52\cdot3^{n-1}$. Khi đó $\dfrac{u_n}{3^n}=\dfrac1{2\cdot3^n}-\dfrac56\to-\dfrac56\approx-0{,}8$.}
\end{ex}
