\begin{ex}
Tìm $\lim u_n$ biết $u_n=\dfrac{n\cdot\sqrt{1+3+5+\cdots+(2n-1)}}{2n^2+1}$.
\choice{\True $\dfrac12$}{$+\infty$}{$1$}{$-\infty$}
\loigiai{\sloppy $1+3+\cdots+(2n-1)=n^2$ nên $u_n=\dfrac{n\cdot n}{2n^2+1}=\dfrac{1}{2+\frac1{n^2}}\to\dfrac12$.}
\end{ex}
