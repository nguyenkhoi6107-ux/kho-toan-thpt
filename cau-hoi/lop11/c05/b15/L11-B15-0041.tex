\begin{ex}
Giới hạn $\lim\left[(2-3n)^4(n+1)^3\right]$ là
\choice{$-\infty$}{\True $+\infty$}{$81$}{$2$}
\loigiai{\sloppy $(2-3n)^4(n+1)^3=n^7\left(\frac2n-3\right)^4\left(1+\frac1n\right)^3$, $n^7\to+\infty$ và thừa số còn lại $\to81$, nên giới hạn bằng $+\infty$.}
\end{ex}
