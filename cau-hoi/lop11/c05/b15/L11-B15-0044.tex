\begin{ex}
Giới hạn $\lim\left(\sqrt{n^2-3n+1}-n\right)$ bằng
\choice{$-3$}{$+\infty$}{$0$}{\True $-\dfrac32$}
\loigiai{\sloppy \[\sqrt{n^2-3n+1}-n=\dfrac{-3n+1}{\sqrt{n^2-3n+1}+n}=\dfrac{-3+\frac1n}{\sqrt{1-\frac3n+\frac1{n^2}}+1}\to-\dfrac32.\]}
\end{ex}
