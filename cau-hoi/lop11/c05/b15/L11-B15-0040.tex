\begin{ex}
Tìm $\lim\left(\dfrac1{n^2}+\dfrac2{n^2}+\cdots+\dfrac n{n^2}\right)$.
\choice{$+\infty$}{\True $\dfrac12$}{$\dfrac1n$}{$0$}
\loigiai{\sloppy $\dfrac{1+2+\cdots+n}{n^2}=\dfrac{n(n+1)}{2n^2}=\dfrac{1+\frac1n}{2}\to\dfrac12$.}
\end{ex}
