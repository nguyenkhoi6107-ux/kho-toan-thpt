\begin{ex}
Cho giới hạn $\lim\limits_{x\to2}\dfrac{x^2-3x+2}{x^2-4}=\dfrac ab$ trong đó $\dfrac ab$ là phân số tối giản. Tính $S=a^2+b^2$.
\choice{$S=20$}{\True $S=17$}{$S=10$}{$S=25$}
\loigiai{\sloppy $\dfrac{(x-1)(x-2)}{(x-2)(x+2)}=\dfrac{x-1}{x+2}\to\dfrac14$, nên $a=1$, $b=4$ và $S=17$.}
\end{ex}
