\begin{ex}
Tính các giới hạn sau.

a) $\lim\limits_{x\to0}\left[x\left(1-\dfrac1x\right)\right]$;

b) $\lim\limits_{x\to+\infty}x\left(\sqrt{x^2+5x+4}-\sqrt{x^2+5x-2}\right)$.
\loigiai{\sloppy a) $x\left(1-\dfrac1x\right)=x-1\to-1$.

b) \[\begin{aligned}x\left(\sqrt{x^2+5x+4}-\sqrt{x^2+5x-2}\right)&=\dfrac{6x}{\sqrt{x^2+5x+4}+\sqrt{x^2+5x-2}}\\&=\dfrac6{\sqrt{1+\frac5x+\frac4{x^2}}+\sqrt{1+\frac5x-\frac2{x^2}}}\to3.\end{aligned}\]}
\end{ex}
