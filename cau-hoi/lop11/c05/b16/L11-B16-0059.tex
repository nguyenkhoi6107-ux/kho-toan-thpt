\begin{ex}
Biết $\lim\limits_{x\to+\infty}\left(\sqrt{4x^2-3x+1}-(ax+b)\right)=0$. Khi đó
\choiceTF
{$a$ là số lẻ.}
{$b>0$.}
{\True $a\cdot b<0$.}
{\True $a-4b=5$.}
\loigiai{\sloppy Tương tự bài trên: $a=2$, $b=-\dfrac34$.
a) $a=2$ là số chẵn. Sai.
b) $b=-\dfrac34<0$. Sai.
c) $ab=-\dfrac32<0$. Đúng.
d) $a-4b=2+3=5$. Đúng.}
\end{ex}
