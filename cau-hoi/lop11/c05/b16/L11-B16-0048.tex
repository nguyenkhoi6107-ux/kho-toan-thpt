\begin{ex}
Tính giới hạn $L=\lim\limits_{x\to-1}\dfrac{x^2-x-2}{3x^2+8x+5}$.
\choice{\True $L=-\dfrac32$}{$L=\dfrac12$}{$L=-\infty$}{$L=0$}
\loigiai{\sloppy $\dfrac{(x+1)(x-2)}{(x+1)(3x+5)}=\dfrac{x-2}{3x+5}\to\dfrac{-3}2$.}
\end{ex}
