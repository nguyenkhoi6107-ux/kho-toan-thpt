\begin{ex}
Tìm giới hạn $I=\lim\limits_{x\to+\infty}\left(x+1-\sqrt{x^2-x+2}\right)$.
\choice{$I=\dfrac12$}{$I=\dfrac{46}{31}$}{$I=\dfrac{17}{11}$}{\True $I=\dfrac32$}
\loigiai{\sloppy \[\begin{aligned}x+1-\sqrt{x^2-x+2}=\dfrac{(x+1)^2-(x^2-x+2)}{x+1+\sqrt{x^2-x+2}}\\ &=\dfrac{3x-1}{x+1+\sqrt{x^2-x+2}}\\ &=\dfrac{3-\frac1x}{1+\frac1x+\sqrt{1-\frac1x+\frac2{x^2}}}\to\dfrac32.\end{aligned}\]}
\end{ex}
