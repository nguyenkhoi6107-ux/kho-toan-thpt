\begin{ex}
Tính giới hạn $\lim\limits_{x\to-\infty}\left(2x^3-x^2+1\right)$.
\choice{$+\infty$}{\True $-\infty$}{$2$}{$0$}
\loigiai{\sloppy $2x^3-x^2+1=x^3\left(2-\frac1x+\frac1{x^3}\right)$ với $x^3\to-\infty$ và thừa số $\to2>0$, nên giới hạn bằng $-\infty$.}
\end{ex}
