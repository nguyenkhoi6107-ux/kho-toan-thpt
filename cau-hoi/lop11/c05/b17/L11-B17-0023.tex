\begin{ex}
Cho hàm số $y=\begin{cases}\dfrac{1-x^3}{1-x}&\text{khi }x<1\\1&\text{khi }x\ge1\end{cases}$. Hãy chọn kết luận \textbf{đúng}.
\choice{\True $y$ liên tục phải tại $x=1$.}{$y$ liên tục tại $x=1$.}{$y$ liên tục trái tại $x=1$.}{$y$ liên tục trên $\mathbb R$.}
\loigiai{\sloppy $f(1)=1$. $\lim\limits_{x\to1^+}f(x)=1=f(1)$ nên liên tục phải tại $1$. $\lim\limits_{x\to1^-}\dfrac{1-x^3}{1-x}=\lim(1+x+x^2)=3\ne f(1)$ nên không liên tục trái, do đó gián đoạn tại $1$.}
\end{ex}
