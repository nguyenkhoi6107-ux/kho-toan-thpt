\begin{ex}
Tìm tham số $m$ để hàm số $f(x)=\begin{cases}x^2+2x-m&\text{khi }x\ne2\\x+m&\text{khi }x=2\end{cases}$ liên tục tại điểm $x_0=2$.
\loigiai{\sloppy $f(2)=2+m$; $\lim\limits_{x\to2}(x^2+2x-m)=8-m$. Liên tục khi $8-m=2+m\Leftrightarrow m=3$.}
\end{ex}
