\begin{ex}
Để hàm số $f(x)=\begin{cases}\dfrac{2x^2-3x+1}{2(x-1)}&\text{khi }x\ne1\\m&\text{khi }x=1\end{cases}$ liên tục tại $x=1$ thì giá trị $m$ bằng
\choice{\True $0{,}5$.}{$1{,}5$.}{$1$.}{$2$.}
\loigiai{\sloppy $\dfrac{(2x-1)(x-1)}{2(x-1)}=\dfrac{2x-1}2\to\dfrac12$; liên tục khi $m=0{,}5$.}
\end{ex}
