\begin{ex}
Xét tính liên tục của hàm số $f(x)=\begin{cases}\dfrac{x^2-6x+5}{x^2-1}&\text{khi }x\ne1\\-2&\text{khi }x=1\end{cases}$ tại điểm $x_0=1$.
\loigiai{\sloppy $\lim\limits_{x\to1}\dfrac{(x-1)(x-5)}{(x-1)(x+1)}=\lim\dfrac{x-5}{x+1}=-2=f(1)$. Hàm số liên tục tại $x_0=1$.}
\end{ex}
