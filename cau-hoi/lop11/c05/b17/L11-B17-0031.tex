\begin{ex}
Tìm $m$ để hàm số $f(x)=\begin{cases}\dfrac{x^2-4}{x+2}&\text{khi }x\ne-2\\m&\text{khi }x=-2\end{cases}$ liên tục tại $x=-2$.
\choice{\True $m=-4$.}{$m=2$.}{$m=4$.}{$m=0$.}
\loigiai{\sloppy $\lim\limits_{x\to-2}\dfrac{(x-2)(x+2)}{x+2}=-4$; liên tục khi $m=f(-2)=-4$.}
\end{ex}
