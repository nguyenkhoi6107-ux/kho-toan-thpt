\begin{ex}
Tìm các giá trị của tham số $m$ để hàm số $f(x)=\begin{cases}\dfrac{x^2-3x+2}{x^2-2x}&\text{khi }x<2\\mx+m+1&\text{khi }x\ge2\end{cases}$ liên tục tại điểm $x=2$.
\choice{$m=\dfrac16$.}{\True $m=-\dfrac16$.}{$m=-\dfrac12$.}{$m=\dfrac12$.}
\loigiai{\sloppy $\lim\limits_{x\to2^-}\dfrac{(x-1)(x-2)}{x(x-2)}=\dfrac12$; $f(2)=3m+1$. Liên tục khi $3m+1=\dfrac12\Leftrightarrow m=-\dfrac16$.}
\end{ex}
