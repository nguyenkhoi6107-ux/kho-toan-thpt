\begin{ex}
Cho hàm số $f(x)=\begin{cases}\dfrac{\sqrt x-1}{x-1}&\text{khi }x\ne1\\a&\text{khi }x=1\end{cases}$. Tìm $a$ để hàm số liên tục tại $x_0=1$.
\choice{$a=0$.}{$a=-\dfrac12$.}{\True $a=\dfrac12$.}{$a=1$.}
\loigiai{\sloppy $\lim\limits_{x\to1}\dfrac{\sqrt x-1}{x-1}=\lim\dfrac1{\sqrt x+1}=\dfrac12$; liên tục khi $a=\dfrac12$.}
\end{ex}
