\begin{ex}
Tìm tất cả các giá trị của tham số $m$ để hàm số $f(x)=\begin{cases}\dfrac{x^2-2x}{x-2}&\text{khi }x>2\\mx-4&\text{khi }x\le2\end{cases}$ liên tục tại $x=2$.
\choice{\True $m=3$.}{$m=2$.}{$m=-2$.}{Không tồn tại $m$.}
\loigiai{\sloppy $\lim\limits_{x\to2^+}\dfrac{x(x-2)}{x-2}=2$; $f(2)=2m-4$. Liên tục khi $2m-4=2\Leftrightarrow m=3$.}
\end{ex}
