\begin{ex}
Tìm $a$ để hàm số $f(x)=\begin{cases}\dfrac{x^2-1}{x-1}&\text{khi }x\ne1\\a&\text{khi }x=1\end{cases}$ liên tục tại điểm $x_0=1$.
\choice{$a=1$.}{$a=0$.}{\True $a=2$.}{$a=-1$.}
\loigiai{\sloppy $\lim\limits_{x\to1}\dfrac{x^2-1}{x-1}=\lim(x+1)=2$; liên tục khi $a=2$.}
\end{ex}
