\begin{ex}
Tìm giá trị của tham số $a$ để hàm số $f(x)=\begin{cases}\dfrac{\sqrt x-1}{x-1}&\text{khi }x>1\\ax-\dfrac12&\text{khi }x\le1\end{cases}$ liên tục tại điểm $x=1$.
\shortans{1}
\loigiai{\sloppy $\lim\limits_{x\to1^+}\dfrac1{\sqrt x+1}=\dfrac12$; $f(1)=a-\dfrac12$. Liên tục khi $a-\dfrac12=\dfrac12\Leftrightarrow a=1$.}
\end{ex}
