\begin{ex}
Tìm $P$ để hàm số $y=\begin{cases}\dfrac{x^2-4x+3}{x-1}&\text{khi }x>1\\6Px-3&\text{khi }x\le1\end{cases}$ liên tục trên $\mathbb R$.
\choice{$P=\dfrac56$.}{$P=\dfrac12$.}{\True $P=\dfrac16$.}{$P=\dfrac13$.}
\loigiai{\sloppy $\lim\limits_{x\to1^+}\dfrac{(x-1)(x-3)}{x-1}=-2$; $f(1)=6P-3$. Liên tục khi $6P-3=-2\Leftrightarrow P=\dfrac16$.}
\end{ex}
