\begin{ex}
Tìm $m$ để hàm số $f(x)=\begin{cases}\dfrac{x^2-16}{x-4}&\text{khi }x>4\\mx+1&\text{khi }x\le4\end{cases}$ liên tục tại điểm $x=4$.
\choice{\True $m=\dfrac74$.}{$m=8$.}{$m=-\dfrac74$.}{$m=-8$.}
\loigiai{\sloppy $\lim\limits_{x\to4^+}\dfrac{(x-4)(x+4)}{x-4}=8$; $f(4)=4m+1$. Liên tục khi $4m+1=8\Leftrightarrow m=\dfrac74$.}
\end{ex}
