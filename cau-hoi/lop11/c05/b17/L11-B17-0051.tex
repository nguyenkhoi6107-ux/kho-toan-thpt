\begin{ex}
Tìm $m$ để hàm số $f(x)=\begin{cases}\dfrac{x^2+4x+3}{x+1}&\text{khi }x>-1\\mx+2&\text{khi }x\le-1\end{cases}$ liên tục tại điểm $x=-1$.
\choice{$m=2$.}{\True $m=0$.}{$m=-4$.}{$m=4$.}
\loigiai{\sloppy $\lim\limits_{x\to-1^+}\dfrac{(x+1)(x+3)}{x+1}=2$; $f(-1)=-m+2$. Liên tục khi $-m+2=2\Leftrightarrow m=0$.}
\end{ex}
