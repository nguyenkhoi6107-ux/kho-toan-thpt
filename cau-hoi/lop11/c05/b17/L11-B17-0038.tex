\begin{ex}
Cho hàm số $f(x)=\begin{cases}\dfrac{x^2-3x+2}{x-2}&\text{khi }x\ne2\\a&\text{khi }x=2\end{cases}$. Hàm số liên tục tại $x=2$ khi $a$ bằng
\choice{\True $1$.}{$0$.}{$2$.}{$-1$.}
\loigiai{\sloppy $\lim\limits_{x\to2}\dfrac{(x-1)(x-2)}{x-2}=1$; liên tục khi $a=1$.}
\end{ex}
