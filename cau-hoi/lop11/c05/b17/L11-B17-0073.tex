\begin{ex}
Tìm $m$ để hàm số $f(x)=\begin{cases}\dfrac{x^3-x^2+2x-2}{x-1}&\text{khi }x\ne1\\3x+m&\text{khi }x=1\end{cases}$ liên tục tại điểm $x=1$.
\shortans{0}
\loigiai{\sloppy $x^3-x^2+2x-2=(x-1)(x^2+2)$ nên $\lim\limits_{x\to1}f(x)=3$. $f(1)=3+m$. Liên tục khi $3+m=3\Leftrightarrow m=0$.}
\end{ex}
