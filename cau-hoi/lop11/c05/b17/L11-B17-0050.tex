\begin{ex}
Tìm $a$ để hàm số $f(x)=\begin{cases}2x+a&\text{khi }x\le1\\\dfrac{x^3-x^2+2x-2}{x-1}&\text{khi }x>1\end{cases}$ liên tục trên $\mathbb R$.
\choice{$a=-2$.}{\True $a=1$.}{$a=2$.}{$a=-1$.}
\loigiai{\sloppy Với $x\ne1$ hàm số liên tục. Tại $x=1$: $f(1)=2+a$, $\lim\limits_{x\to1^-}f(x)=2+a$, $\lim\limits_{x\to1^+}\dfrac{(x-1)(x^2+2)}{x-1}=3$. Liên tục khi $2+a=3\Leftrightarrow a=1$.}
\end{ex}
