\begin{ex}
Cho $\tan\alpha=2$. Tính $\tan\left(\alpha+\dfrac{\pi}{4}\right)$.
\choice{$-\dfrac13$}{$\dfrac13$}{\True $-3$}{$1$}
\loigiai{$\tan\left(\alpha+\dfrac\pi4\right)=\dfrac{\tan\alpha+1}{1-\tan\alpha}=\dfrac{3}{-1}=-3$.}
\end{ex}
