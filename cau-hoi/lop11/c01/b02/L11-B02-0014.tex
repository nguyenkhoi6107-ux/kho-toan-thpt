\begin{ex}
Cho $\tan\alpha=3$. Hãy tính giá trị của biểu thức $\tan\left(\alpha-\dfrac{\pi}{3}\right)$.
\choice{$\dfrac{3+\sqrt3}{1-3\sqrt3}$}{\True $\dfrac{3-\sqrt3}{1+3\sqrt3}$}{$\dfrac{1-\sqrt3}{1+\sqrt3}$}{$\dfrac{1-\sqrt3}{1+3\sqrt3}$}
\loigiai{$\tan\left(\alpha-\dfrac\pi3\right)=\dfrac{\tan\alpha-\sqrt3}{1+\sqrt3\tan\alpha}=\dfrac{3-\sqrt3}{1+3\sqrt3}$.}
\end{ex}
