\begin{ex}
Khẳng định nào sau đây \textbf{đúng}?
\choice{$\tan\alpha+\cot\alpha=1$ $\left(\alpha\ne\dfrac{k\pi}{2},\ k\in\mathbb Z\right)$}{\True $\dfrac{1}{\cos^2\alpha}=1+\tan^2\alpha$ $\left(\alpha\ne\dfrac{\pi}{2}+k\pi,\ k\in\mathbb Z\right)$}{$\dfrac{1}{\cot^2\alpha}=1+\sin^2\alpha$ $\left(\alpha\ne\dfrac{k\pi}{2},\ k\in\mathbb Z\right)$}{$\sin^2a+\cos^2b=1$}
\loigiai{Với $\cos\alpha\ne0$: $1+\tan^2\alpha=\dfrac{\cos^2\alpha+\sin^2\alpha}{\cos^2\alpha}=\dfrac{1}{\cos^2\alpha}$. Các đẳng thức còn lại sai (ví dụ $\tan\alpha\cdot\cot\alpha=1$ chứ không phải tổng).}
\end{ex}
