\choice{$x=\dfrac{\pi}{2}+k\pi$ $(k\in\mathbb Z)$}{$x=\pi+k2\pi$ $(k\in\mathbb Z)$}{\True $x=k\pi$ $(k\in\mathbb Z)$}{$x=k2\pi$ $(k\in\mathbb Z)$}

\end{ex}
