\begin{ex}

\choice{$x=-\dfrac{\pi}{2}+k\pi,\ k\in\mathbb Z$}{\True $x=-\dfrac{\pi}{2}+k2\pi,\ k\in\mathbb Z$}{$x=k\pi,\ k\in\mathbb Z$}{$x=\dfrac{3\pi}{2}+k\pi,\ k\in\mathbb Z$}
