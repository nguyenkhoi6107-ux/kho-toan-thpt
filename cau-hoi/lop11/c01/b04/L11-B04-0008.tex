\begin{ex}
Tìm số nghiệm thuộc đoạn $[\pi;2\pi]$ của phương trình $2\sin\left(x+\dfrac{\pi}{3}\right)=0$.
\choice{$0$}{\True $1$}{$2$}{$3$}
\loigiai{$x+\dfrac\pi3=k\pi\Leftrightarrow x=-\dfrac\pi3+k\pi$. Trên $[\pi;2\pi]$ chỉ có $x=\dfrac{5\pi}{3}$ (ứng với $k=2$). Có $1$ nghiệm.}
\end{ex}
