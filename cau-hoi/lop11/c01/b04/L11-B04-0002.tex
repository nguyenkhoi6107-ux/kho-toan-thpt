\begin{ex}
Phương trình $2\cos2x+1=0$ có nghiệm là
\choice{$x=-\dfrac\pi3+2k\pi$ và $x=\dfrac\pi3+2k\pi$ $(k\in\mathbb Z)$}{\True $x=-\dfrac\pi3+k\pi$ và $x=\dfrac\pi3+k\pi$ $(k\in\mathbb Z)$}{$x=-\dfrac{2\pi}3+2k\pi$ và $x=\dfrac{2\pi}3+2k\pi$ $(k\in\mathbb Z)$}{$x=-\dfrac\pi6+k\pi$ và $x=\dfrac\pi6+k\pi$ $(k\in\mathbb Z)$}
\loigiai{$2\cos2x+1=0\Leftrightarrow\cos2x=-\dfrac12=\cos\dfrac{2\pi}3\Leftrightarrow2x=\pm\dfrac{2\pi}3+k2\pi\Leftrightarrow x=\pm\dfrac\pi3+k\pi$ $(k\in\mathbb Z)$.}
\end{ex}
