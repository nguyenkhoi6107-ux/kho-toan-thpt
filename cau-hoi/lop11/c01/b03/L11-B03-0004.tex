\begin{ex}
Hàm số $y=\sin2x$ có chu kì tuần hoàn là
\choice{\True $T=\pi$}{$T=\dfrac{\pi}{2}$}{$T=4\pi$}{$T=2\pi$}
\loigiai{Hàm số $y=\sin(ax)$ tuần hoàn với chu kì $T=\dfrac{2\pi}{|a|}=\dfrac{2\pi}{2}=\pi$.}
\end{ex}
