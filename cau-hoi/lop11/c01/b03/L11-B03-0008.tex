\begin{ex}
Tập xác định của hàm số $y=\dfrac{\sin x-1}{\tan x}$ là
\choice{$D=\mathbb R\setminus\left\{\dfrac{\pi}{2}+k\pi,\ k\in\mathbb Z\right\}$}{\True $D=\mathbb R\setminus\left\{\dfrac{k\pi}{2},\ k\in\mathbb Z\right\}$}{$D=\mathbb R\setminus\{k\pi,\ k\in\mathbb Z\}$}{$D=\mathbb R$}
\loigiai{Cần $\tan x$ xác định và khác $0$: $x\ne\dfrac\pi2+k\pi$ và $x\ne k\pi$, tức là $x\ne\dfrac{k\pi}{2}$ $(k\in\mathbb Z)$.}
\end{ex}
