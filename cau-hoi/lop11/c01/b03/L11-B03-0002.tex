\begin{ex}
Tập xác định của hàm số $y=\tan\left(x+\dfrac{\pi}{3}\right)$ là
\choice{$D=\mathbb R\setminus\left\{\dfrac{\pi}{2}+k\pi,\ k\in\mathbb Z\right\}$}{\True $D=\mathbb R\setminus\left\{\dfrac{\pi}{6}+k\pi,\ k\in\mathbb Z\right\}$}{$D=\mathbb R\setminus\left\{\dfrac{\pi}{3}+k\pi,\ k\in\mathbb Z\right\}$}{$D=\mathbb R\setminus\left\{-\dfrac{\pi}{3}+k\pi,\ k\in\mathbb Z\right\}$}
\loigiai{Hàm số xác định khi $\cos\left(x+\dfrac{\pi}3\right)\ne0$, tức $x+\dfrac{\pi}3\ne\dfrac{\pi}2+k\pi$, hay $x\ne\dfrac{\pi}6+k\pi$ $(k\in\mathbb Z)$. Chọn B.}
\end{ex}
