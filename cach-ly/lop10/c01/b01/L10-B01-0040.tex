Cho mệnh đề chứa biến $P(x)$, xét tính đúng sai của các mệnh đề sau.
\choiceTF
{$P(1)$.}
{$P\left(\dfrac13\right)$.}
{$\forall x\in\mathbb N,\ P(x)$.}
{$\exists x\in\mathbb N,\ P(x)$.}
\loigiai{Không giải được vì thiếu biểu thức của $P(x)$.}
\end{ex}
